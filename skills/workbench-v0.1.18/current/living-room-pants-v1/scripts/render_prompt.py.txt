"""Render the two frozen prompt templates from explicitly supplied product fields. Standard library only."""
import argparse
import json
from pathlib import Path
import re

TOKEN = re.compile(r"\{\{([A-Z][A-Z0-9_]*)\}\}")

def render(skill_root, values):
    templates = [(skill_root / "references" / f"clip-{i}-template.txt").read_text(encoding="utf-8") for i in (1, 2)]
    expected = {k for t in templates for k in TOKEN.findall(t)}
    if not isinstance(values, dict) or set(values) != expected:
        actual = set(values) if isinstance(values, dict) else set()
        raise ValueError(f"Field mismatch. Missing: {sorted(expected-actual)}; unexpected: {sorted(actual-expected)}")
    for k, value in values.items():
        if not isinstance(value, str) or not value.strip():
            raise ValueError(f"{k} must be a nonempty string")
        if "{{" in value or "}}" in value:
            raise ValueError(f"{k} contains unresolved template markers")
    rendered = [TOKEN.sub(lambda m: values[m.group(1)], t) for t in templates]
    return rendered

def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--values", required=True, type=Path)
    p.add_argument("--output", required=True, type=Path)
    args = p.parse_args()
    values = json.loads(args.values.read_text(encoding="utf-8-sig"))
    root = Path(__file__).resolve().parents[1]
    parts = render(root, values)
    dest = args.output.resolve()
    if dest.exists():
        raise FileExistsError(f"Refusing to overwrite: {dest}")
    dest.parent.mkdir(parents=True, exist_ok=True)
    text = "两段分别提交，每段只需本次产品图和对应完整提示词。\n\n"
    for i, part in enumerate(parts, 1):
        text += f"【第{i}段】\n\n" + part + "\n\n"
    with dest.open("x", encoding="utf-8", newline="") as handle:
        handle.write(text)
    print(str(dest))

if __name__ == "__main__":
    main()
